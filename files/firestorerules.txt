rules_version = '2';

// ⚠️ 請將此規則貼到 Firebase Console -> Firestore Database -> 規則(Rules)，
//    貼上後按「發布」，否則任何人只要拿到你的網站原始碼（GitHub Pages 是公開的），
//    就能直接連到你的 Firestore 資料庫任意讀寫／刪除資料。
//
// 設計原則：
// 1. quotes（估價單）維持公開讀寫，比照原本雛形「業務不用登入就能開單、查單」的使用方式。
//    如果之後想連估價單也鎖起來，把下面 quotes 區塊的 allow 規則改成
//    `allow read, write: if request.auth != null;` 即可。
// 2. equipment（客戶儀器保養/校正紀錄）與 settings 的「寫入」都需要登入
//    （對應網站上「儀器管理系統」「管理員後台」的登入關卡）。
// 3. settings（業務名單、價格表）維持「讀取」公開，因為估價單頁面在沒登入時
//    也需要讀取業務下拉選單與價格建議。

service cloud.firestore {
  match /databases/{database}/documents {

    match /quotes/{quoteId} {
      allow read, write: if true;
    }

    match /equipment/{equipmentId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
